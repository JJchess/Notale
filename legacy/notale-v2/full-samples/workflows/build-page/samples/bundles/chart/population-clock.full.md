const $=s=>document.querySelector(s);let rows=[],options=[],index=0,live=true,last='',ready=false;
export function getOptions(data,h,m){const num=Number(`${Number(h)}${String(m).padStart(2,'0')}`),all=data.filter(d=>Number(d.population)===num),notTownship=all.filter(d=>d.type!=='s'),notUn=notTownship.filter(d=>d.type!=='u');return notUn.length?notUn:notTownship.length?notTownship:all}
function mapUrl(d){return 'https://www.google.com/maps/place/?q='+[d.name,d.county,d.state].map(s=>s.replace(/ /g,'+')).join('+')}
function show(){const d=options[index];$('#place').textContent=d.name;$('#place').href=mapUrl(d);$('#state').textContent=', '+d.state;$('#count').textContent=options.length;$('#place-word').textContent=options.length===1?'place':'places';$('#next').disabled=$('#previous').disabled=options.length<2;$('#list-title').textContent=`${d.population} 人 · ${options.length} 座候选城镇`;
 const list=$('#places');list.replaceChildren();options.forEach((d,i)=>{const li=document.createElement('li'),b=document.createElement('button'),small=document.createElement('small');b.textContent=`${d.name}, ${d.state}`;small.textContent=`${d.county} · ${d.population} 人`;b.append(small);b.onclick=()=>{index=i;show();$('#matches').close()};li.append(b);list.append(li)});
}
function setTime(value,isLive=false){const [hour,minute]=value.split(':').map(Number),h=hour%12||12,m=String(minute).padStart(2,'0');live=isLive;last=value;$('#time').value=value;$('#digits').textContent=`${h}${m}`;$('#period').textContent=hour<12?'AM':'PM';$('#live').setAttribute('aria-pressed',String(live));$('#mode').textContent=live?'实时时钟 · 使用设备本地时间':'手动时间 · 可继续选时，或返回实时时钟';options=getOptions(rows,h,m);index=Math.floor(Math.random()*options.length);show()}
function now(){const d=new Date();return `${String(d.getHours()).padStart(2,'0')}:${String(d.getMinutes()).padStart(2,'0')}`}
$('#next').onclick=()=>{index=(index+1)%options.length;show()};$('#previous').onclick=()=>{index=(index+options.length-1)%options.length;show()};$('#time').oninput=e=>{if(e.target.value)setTime(e.target.value)};$('#live').onclick=()=>setTime(now(),true);$('#theme').onclick=()=>{const light=document.body.classList.toggle('light');$('#theme').textContent=light?'深色模式':'浅色模式';$('#theme').setAttribute('aria-pressed',String(light))};$('#details').onclick=()=>$('#matches').showModal();$('#matches').onclick=e=>{if(e.target===$('#matches')){const r=e.target.getBoundingClientRect();if(e.clientX<r.left||e.clientX>r.right||e.clientY<r.top||e.clientY>r.bottom)e.target.close()}};
try{rows=await fetch('assets/populations.json').then(r=>{if(!r.ok)throw Error();return r.json()});const query=new URLSearchParams(location.search),time=query.get('time');document.querySelectorAll('button,input').forEach(b=>b.disabled=false);if(time&&/^\d{1,2}-[0-5]\d$/.test(time)&&Number(time.split('-')[0])>=1&&Number(time.split('-')[0])<=12){const [h,m]=time.split('-');const hour=Number(h)%12+(query.get('period')==='PM'?12:0);setTime(`${String(hour).padStart(2,'0')}:${m}`)}else setTime(now(),true);ready=true;setInterval(()=>{if(live&&now()!==last)setTime(now(),true)},250)}catch{$('#mode').textContent='数据载入失败，请刷新重试。'}
window.populationClock={getOptions,setTime,get state(){return {ready,live,index,options,rows,time:last}}};
