// Data and image composition: Intro.SummaryBottles.svelte + ChartScroll.SummaryBottles.svelte.
let data,metric='medianPrice',selected=null;const rack=document.querySelector('#rack');const format=(v)=>metric==='medianPrice'?'$'+v.toFixed(2):metric==='medianRating'?Number(v.toFixed(2)).toString():v.toFixed(1)+'%';
function render(){if(!data)return;const previousFocus=document.activeElement;const mobile=innerWidth<601,rows=[...data.groups].sort((a,b)=>a[metric]-b[metric]||a.animal.localeCompare(b.animal));rows.forEach((d,i)=>{const el=[...rack.children].find(el=>el.dataset.animal===d.animal);rack.append(el);el.style.transform=`translate(${mobile?0:i*rack.clientWidth/16}px,${mobile?i*66:selected===d.animal?-13:0}px)`;el.setAttribute('aria-pressed',selected===d.animal);el.dataset.rank=i+1;el.querySelector('.value').textContent=format(d[metric]);el.querySelector('.silhouette').src=`assets/blank-bottle${selected===d.animal?'-outline':''}${mobile?'-horiz':''}.png`;el.setAttribute('aria-label',`${d.animal}, ${format(d[metric])}, rank ${i+1}; track this group`)});document.querySelectorAll('nav button').forEach(b=>b.setAttribute('aria-pressed',b.dataset.metric===metric));document.querySelector('#tracking').textContent=selected?`Tracking ${selected} · position ${rows.findIndex(d=>d.animal===selected)+1} of 16 · ${format(rows.find(d=>d.animal===selected)[metric])}`:'Select a bottle to track it across rankings.';document.querySelector('#benchmarks').textContent=`Reference: animal wines ${format(data.animal[metric])} · all wines ${format(data.all[metric])}`;rack.dataset.metric=metric;if(previousFocus&&previousFocus!==document.body)previousFocus.focus({preventScroll:true});}
fetch('data.json').then(r=>r.json()).then(d=>{data=d;data.groups.forEach(g=>{const el=document.createElement('button');el.className='bottle';el.dataset.animal=g.animal;el.innerHTML=`<span class="value"></span><span class="art"><img class="silhouette" src="assets/blank-bottle.png" alt=""><img class="icon" src="assets/${g.animal.replace(/[^a-zA-Z0-9]/g,'')}.png" alt=""></span><span class="name">${g.animal}</span>`;el.onclick=()=>{selected=selected===g.animal?null:g.animal;render()};rack.append(el)});render()}).catch(()=>document.querySelector('#tracking').textContent='The local data could not load. Please reload.');document.querySelectorAll('nav button').forEach(b=>b.onclick=()=>{metric=b.dataset.metric;render()});window.addEventListener('resize',render);
